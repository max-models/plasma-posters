import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

## What it does

<CardGrid>
	<Card title="Blocks in columns" icon="document">
		A poster is a title, the authors and blocks of text and figures, stacked in columns.
	</Card>
	<Card title="LaTeX inside" icon="pencil">
		Text, titles and captions are LaTeX, so equations look as they do in your papers.
	</Card>
	<Card title="Your figures" icon="seti:image">
		Figures from Matplotlib, plasma-plots or maxplotlib go in as PDF, PNG or JPEG files.
	</Card>
	<Card title="Print-ready PDF" icon="rocket">
		The poster is a tikzposter LaTeX file compiled with lualatex, in A0, A1 or A2.
	</Card>
</CardGrid>
